/**
 * Neetu's Kitchen Nagawara - Main Application JavaScript
 * All interactive features: Live Business Hours, Menu Filter, WhatsApp Order Builder,
 * Lightbox Gallery, FAQ Accordion, Contact Form, Mobile Drawer, and Scroll Animations.

document.addEventListener('DOMContentLoaded', () => {
  initIcons();

/**
 * 1. Initialize Lucide Icons if loaded via CDN
 */
function initIcons() {
  if (typeof window.lucide !== 'undefined' && window.lucide.createIcons) {
    window.lucide.createIcons();
  }
}

/**
 * 2. Mobile Hamburger Navigation Drawer
 */
function initMobileNav() {
  const menuBtn = document.getElementById('mobile-menu-btn');
  const closeBtn = document.getElementById('mobile-menu-close');
  const drawer = document.getElementById('mobile-drawer');
  const overlay = document.getElementById('mobile-overlay');
  const navLinks = document.querySelectorAll('.mobile-nav-link');

  function openDrawer() {
    if (!drawer || !overlay) return;
    drawer.classList.remove('translate-x-full');
    overlay.classList.remove('hidden', 'opacity-0');
    overlay.classList.add('opacity-100');
    document.body.style.overflow = 'hidden';
  }

  function closeDrawer() {
    if (!drawer || !overlay) return;
    drawer.classList.add('translate-x-full');
    overlay.classList.remove('opacity-100');
    overlay.classList.add('opacity-0');
    setTimeout(() => {
      overlay.classList.add('hidden');
    }, 300);
    document.body.style.overflow = '';
  }

  if (menuBtn) menuBtn.addEventListener('click', openDrawer);
  if (closeBtn) closeBtn.addEventListener('click', closeDrawer);
  if (overlay) overlay.addEventListener('click', closeDrawer);

  navLinks.forEach((link) => {
    link.addEventListener('click', closeDrawer);
  });
}

/**
 * 3. Live India Business Hours Indicator
 * Neetu's Kitchen Nagawara: Open Daily 8:00 AM - 10:30 PM (22:30) IST
 */
function initLiveStatusBadge() {
  const badgeEl = document.getElementById('live-status-badge');
  const statusDot = document.getElementById('live-status-dot');
  const statusText = document.getElementById('live-status-text');
  const hoursTableStatus = document.getElementById('hours-live-tag');

  function updateStatus() {
    // Get current India Standard Time (UTC+5:30)
    const now = new Date();
    const utcTime = now.getTime() + now.getTimezoneOffset() * 60000;
    const istTime = new Date(utcTime + 3600000 * 5.5);
    
    const currentHour = istTime.getHours();
    const currentMin = istTime.getMinutes();
    const timeInMinutes = currentHour * 60 + currentMin;

    // Open from 8:00 (480 mins) to 22:30 (1350 mins)
    const openTime = 8 * 60;
    const closeTime = 22 * 60 + 30;
    const isOpen = timeInMinutes >= openTime && timeInMinutes <= closeTime;

    if (isOpen) {
      if (statusDot) {
        statusDot.className = 'w-2.5 h-2.5 rounded-full bg-emerald-500 animate-pulse';
      }
      if (statusText) {
        statusText.textContent = 'OPEN NOW • Closes 10:30 PM';
        statusText.className = 'text-xs font-semibold text-emerald-700 tracking-wide';
      }
      if (badgeEl) {
        badgeEl.className = 'inline-flex items-center gap-2 px-3 py-1.5 rounded-full bg-emerald-50 border border-emerald-200/80 shadow-sm';
      }
      if (hoursTableStatus) {
        hoursTableStatus.textContent = 'OPEN NOW';
        hoursTableStatus.className = 'ml-2 px-2.5 py-0.5 rounded-full text-xs font-bold bg-emerald-100 text-emerald-800';
      }
    } else {
      if (statusDot) {
        statusDot.className = 'w-2.5 h-2.5 rounded-full bg-rose-500';
      }
      if (statusText) {
        statusText.textContent = 'CLOSED NOW • Opens 8:00 AM';
        statusText.className = 'text-xs font-semibold text-rose-700 tracking-wide';
      }
      if (badgeEl) {
        badgeEl.className = 'inline-flex items-center gap-2 px-3 py-1.5 rounded-full bg-rose-50 border border-rose-200/80 shadow-sm';
      }
      if (hoursTableStatus) {
        hoursTableStatus.textContent = 'CLOSED NOW';
        hoursTableStatus.className = 'ml-2 px-2.5 py-0.5 rounded-full text-xs font-bold bg-rose-100 text-rose-800';
      }
    }
  }

  updateStatus();
  setInterval(updateStatus, 60000);
}

/**
 * 4. Interactive Menu Category Filtering
 */
function initMenuFilters() {
  const tabs = document.querySelectorAll('.menu-tab');
  const items = document.querySelectorAll('.menu-item-card');
  const emptyState = document.getElementById('menu-empty-state');

  tabs.forEach((tab) => {
    tab.addEventListener('click', () => {
      const category = tab.getAttribute('data-category');

      // Update active button state
      tabs.forEach((t) => t.classList.remove('active', 'bg-red-600', 'text-white'));
      tabs.forEach((t) => t.classList.add('bg-white', 'text-slate-700', 'hover:bg-slate-50'));
      tab.classList.add('active');
      tab.classList.remove('bg-white', 'text-slate-700', 'hover:bg-slate-50');

      let visibleCount = 0;
      items.forEach((item) => {
        const itemCat = item.getAttribute('data-category');
        if (category === 'all' || itemCat === category) {
          item.style.display = 'flex';
          setTimeout(() => {
            item.style.opacity = '1';
            item.style.transform = 'scale(1)';
          }, 50);
          visibleCount++;
        } else {
          item.style.opacity = '0';
          item.style.transform = 'scale(0.96)';
          setTimeout(() => {
            item.style.display = 'none';
          }, 150);
        }
      });

      if (emptyState) {
        emptyState.style.display = visibleCount === 0 ? 'block' : 'none';
      }
    });
  });
}

/**
 * 5. Direct WhatsApp Order Button Handler
 */
function initWhatsAppOrderButtons() {
  const phone = '919886012345'; // Neetu's Kitchen Nagawara WhatsApp number
  const orderBtns = document.querySelectorAll('.order-wa-btn');

  orderBtns.forEach((btn) => {
    btn.addEventListener('click', (e) => {
      e.preventDefault();
      const itemName = btn.getAttribute('data-item') || 'Delicious Food Item';
      const itemPrice = btn.getAttribute('data-price') || '';
      
      const text = `Hello Neetu's Kitchen Nagawara! 🍲\n\nI would like to order:\n• *${itemName}* (${itemPrice})\n\nPlease let me know the availability and estimated delivery time to my address in Nagawara / Hebbal area. Thank you!`;
      const encoded = encodeURIComponent(text);
      const waUrl = `https://wa.me/${phone}?text=${encoded}`;
      window.open(waUrl, '_blank', 'noopener,noreferrer');
    });
  });

  // Custom Multi-Item WhatsApp Order Builder
  const buildOrderBtn = document.getElementById('open-custom-order-btn');
  const customOrderModal = document.getElementById('custom-order-modal');
  const closeOrderModal = document.getElementById('close-custom-order');
  const sendCustomOrderBtn = document.getElementById('send-custom-order-btn');

  if (buildOrderBtn && customOrderModal) {
    buildOrderBtn.addEventListener('click', () => {
      customOrderModal.classList.remove('hidden');
      document.body.style.overflow = 'hidden';
    });
  }

  if (closeOrderModal && customOrderModal) {
    closeOrderModal.addEventListener('click', () => {
      customOrderModal.classList.add('hidden');
      document.body.style.overflow = '';
    });
  }

  if (sendCustomOrderBtn) {
    sendCustomOrderBtn.addEventListener('click', () => {
      const nameEl = document.getElementById('order-cust-name');
      const addressEl = document.getElementById('order-cust-address');
      const itemsEl = document.getElementById('order-cust-items');
      
      const name = nameEl ? nameEl.value.trim() : '';
      const address = addressEl ? addressEl.value.trim() : '';
      const itemsText = itemsEl ? itemsEl.value.trim() : '';

      if (!itemsText) {
        showToast('Please list the dishes you would like to order.', 'error');
        return;
      }

      const message = `Hello Neetu's Kitchen Nagawara! 🍲\n\n*NEW ORDER REQUEST*\n• *Customer Name*: ${name || 'Valued Customer'}\n• *Delivery Location*: ${address || 'Nagawara / Hebbal area'}\n\n*Order Details*:\n${itemsText}\n\nPlease confirm order acceptance and total amount. Thank you!`;
      const url = `https://wa.me/${phone}?text=${encodeURIComponent(message)}`;
      window.open(url, '_blank', 'noopener,noreferrer');
      
      if (customOrderModal) {
        customOrderModal.classList.add('hidden');
        document.body.style.overflow = '';
      }
      showToast('Redirecting to WhatsApp to complete your order!');
    });
  }
}

/**
 * 6. Photo Gallery Lightbox Modal
 */
function initGalleryLightbox() {
  const modal = document.getElementById('lightbox-modal');
  const modalImg = document.getElementById('lightbox-img');
  const modalTitle = document.getElementById('lightbox-title');
  const modalDesc = document.getElementById('lightbox-desc');
  const closeBtn = document.getElementById('lightbox-close');
  const backdrop = document.getElementById('lightbox-backdrop');

  const galleryItems = document.querySelectorAll('.gallery-item');

  galleryItems.forEach((item) => {
    item.addEventListener('click', () => {
      const src = item.getAttribute('data-img');
      const title = item.getAttribute('data-title');
      const desc = item.getAttribute('data-desc');

      if (modal && modalImg) {
        modalImg.src = src;
        modalImg.alt = title || 'Neetu\'s Kitchen Nagawara Dish';
        if (modalTitle) modalTitle.textContent = title;
        if (modalDesc) modalDesc.textContent = desc;
        modal.classList.remove('hidden');
        document.body.style.overflow = 'hidden';
      }
    });
  });

  function closeLightbox() {
    if (modal) {
      modal.classList.add('hidden');
      document.body.style.overflow = '';
    }
  }

  if (closeBtn) closeBtn.addEventListener('click', closeLightbox);
  if (backdrop) backdrop.addEventListener('click', closeLightbox);

  document.addEventListener('keydown', (e) => {
    if (e.key === 'Escape' && modal && !modal.classList.contains('hidden')) {
      closeLightbox();
    }
  });
}

/**
 * 7. FAQ Accordion Toggle
 */
function initFAQAccordion() {
  const faqItems = document.querySelectorAll('.faq-item');

  faqItems.forEach((item) => {
    const button = item.querySelector('.faq-button');
    if (button) {
      button.addEventListener('click', () => {
        const isActive = item.classList.contains('active');
        
        // Optional: close other open items for neat accordion
        faqItems.forEach((other) => {
          if (other !== item) {
            other.classList.remove('active');
          }
        });

        if (isActive) {
          item.classList.remove('active');
        } else {
          item.classList.add('active');
        }
      });
    }
  });
}

/**
 * 8. Customer Reviews Tag Filtering & Swiggy/Google highlight
 */
function initReviewFilters() {
  const filterBtns = document.querySelectorAll('.review-filter-btn');
  const reviewCards = document.querySelectorAll('.review-card');

  filterBtns.forEach((btn) => {
    btn.addEventListener('click', () => {
      const tag = btn.getAttribute('data-filter');

      filterBtns.forEach((b) => {
        b.classList.remove('bg-red-600', 'text-white', 'shadow-md');
        b.classList.add('bg-white', 'text-slate-700', 'hover:bg-slate-100');
      });
      btn.classList.add('bg-red-600', 'text-white', 'shadow-md');
      btn.classList.remove('bg-white', 'text-slate-700', 'hover:bg-slate-100');

      reviewCards.forEach((card) => {
        const cardTag = card.getAttribute('data-source');
        if (tag === 'all' || cardTag === tag) {
          card.style.display = 'flex';
        } else {
          card.style.display = 'none';
        }
      });
    });
  });
}

/**
 * 9. Contact Form Validation & Toast Notification
 */
function initContactForm() {
  const form = document.getElementById('contact-form');

  if (form) {
    form.addEventListener('submit', (e) => {
      e.preventDefault();

      const nameInput = document.getElementById('contact-name');
      const phoneInput = document.getElementById('contact-phone');
      const serviceInput = document.getElementById('contact-service');
      const msgInput = document.getElementById('contact-message');

      const name = nameInput ? nameInput.value.trim() : '';
      const phone = phoneInput ? phoneInput.value.trim() : '';
      const service = serviceInput ? serviceInput.value : 'General Inquiry';
      const msg = msgInput ? msgInput.value.trim() : '';

      if (!name || !phone) {
        showToast('Please enter your name and contact phone number.', 'error');
        return;
      }

      // Show instant success feedback
      showToast(`Thank you, ${name}! Your inquiry about "${service}" has been received. We will contact you shortly via WhatsApp/Phone.`);
      form.reset();
    });
  }
}

/**
 * 10. Scroll Reveal Animation using Intersection Observer
 */
function initScrollAnimations() {
  const revealElements = document.querySelectorAll('.reveal');

  if ('IntersectionObserver' in window) {
    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {
            entry.target.classList.add('active');
            observer.unobserve(entry.target);
          }
        });
      },
      {
        threshold: 0.1,
        rootMargin: '0px 0px -40px 0px',
      }
    );

    revealElements.forEach((el) => observer.observe(el));
  } else {
    // Fallback if browser doesn't support IntersectionObserver
    revealElements.forEach((el) => el.classList.add('active'));
  }
}

/**
 * Helper: Show Toast Notification
 */
function showToast(message, type = 'success') {
  let toastEl = document.getElementById('toast-notification');

  if (!toastEl) {
    toastEl = document.createElement('div');
    toastEl.id = 'toast-notification';
    toastEl.className =
      'toast-popup fixed bottom-6 left-1/2 -translate-x-1/2 z-50 px-5 py-3.5 rounded-xl shadow-2xl flex items-center gap-3 text-sm font-medium border transition-all duration-300 transform translate-y-20 opacity-0 pointer-events-none max-w-md w-11/12';
    document.body.appendChild(toastEl);
  }

  if (type === 'error') {
    toastEl.className =
      'toast-popup fixed bottom-6 left-1/2 -translate-x-1/2 z-50 px-5 py-3.5 rounded-xl shadow-2xl flex items-center gap-3 text-sm font-medium border border-rose-300 bg-rose-50 text-rose-800 transition-all duration-300 transform translate-y-0 opacity-100 pointer-events-auto max-w-md w-11/12';
    toastEl.innerHTML = `
      <span class="w-2.5 h-2.5 rounded-full bg-rose-500 shrink-0"></span>
      <span>${message}</span>
    `;
  } else {
    toastEl.className =
      'toast-popup fixed bottom-6 left-1/2 -translate-x-1/2 z-50 px-5 py-3.5 rounded-xl shadow-2xl flex items-center gap-3 text-sm font-medium border border-emerald-300 bg-emerald-50 text-emerald-900 transition-all duration-300 transform translate-y-0 opacity-100 pointer-events-auto max-w-md w-11/12';
    toastEl.innerHTML = `
      <span class="w-2.5 h-2.5 rounded-full bg-emerald-500 animate-pulse shrink-0"></span>
      <span>${message}</span>
    `;
  }

  // Auto hide after 4.5 seconds
  setTimeout(() => {
    toastEl.classList.remove('translate-y-0', 'opacity-100', 'pointer-events-auto');
    toastEl.classList.add('translate-y-20', 'opacity-0', 'pointer-events-none');
  }, 4500);
}
